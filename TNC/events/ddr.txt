country_event = { # Congress preparing result
	id = ddrcongress.6
	title = ddrcongress.6.t
	desc = ddrcongress.6.d
	picture = GFX_report_event_generic_diplomacy1

	is_triggered_only = yes

	trigger = {
		OR = {
			has_country_flag = {
				flag = DDR_accelerationist
				value > 1
			}
			has_country_flag = {
				flag = DDR_schizo
				value > 1
			}
			has_country_flag = {
				flag = DDR_thalmann
				value > 1
			}
		}
	}

	option = {
		name = ddrcongress.6.a
		add_political_power = -50
		hidden_effect = {
			if = {
				limit = {
					has_country_flag = {
						flag = DDR_accelerationist
						value > 1
					}
				}
				country_event = { id = ddrcongress.7 days = 20 }
			}
			else_if = {
				limit = {
					has_country_flag = {
						flag = DDR_thalmann
						value > 1
					}
				}
				country_event = { id = ddrcongress.8 days = 20 }
			}
			else_if = {
				limit = {
					has_country_flag = {
						flag = DDR_schizo
						value > 1
					}
				}
				country_event = { id = ddrcongress.9 days = 20 }
			}
		}
	}
}
